cmake_minimum_required(VERSION 3.16)
if(NOT AC_BOARD MATCHES "^(H2|R9)$")
    message(FATAL_ERROR "Select the exact populated board: -DAC_BOARD=H2 or -DAC_BOARD=R9")
endif()
if(NOT AC_SOURCE_SHA256 MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "AC_SOURCE_SHA256 must bind this build to its source manifest")
endif()
string(LENGTH "${AC_SOURCE_SHA256}" AC_HASH_LENGTH)
if(NOT AC_HASH_LENGTH EQUAL 64)
    message(FATAL_ERROR "AC_SOURCE_SHA256 must contain 64 hexadecimal characters")
endif()
add_compile_definitions(AC_BOARD_${AC_BOARD}=1 AC_SOURCE_SHA256="${AC_SOURCE_SHA256}")
set(PROJECT_VER "0.1.0-commissioning.1")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(ac_integrated_commissioning)
